$$
\frac{x^{2}}{\left( x^{2} + 1 \right)\left( x^{2} + 4 \right)} = \frac{A}{\left( x^{2} + 1 \right)} + \frac{B}{\left( x^{2} + 4 \right)}, \quad x^{2} = A\left( x^{2} + 4 \right) + B\left( x^{2} + 1 \right).
$$

Let $x = \sqrt{-1} \quad  or  \quad x^2 = -1,$

$$
\therefore - 1 = 3A , \therefore A = - \frac{1}{3},
$$

$$
x^{2}+\frac{1}{3}\left(x^{2}+4\right)=\mathrm{B}\left(x^{2}+1\right)
$$

$\frac{4\left(x^{2}+1\right)}{3}=\mathrm{B}\left(x^{2}+1\right), \therefore \mathrm{B}=\frac{4}{3}$

$$
\begin{aligned}\int \frac{x^{2}dx}{\left( x^{2} + 1 \right)\left( x^{2} + 4 \right)} &= - \frac{1}{3}\int\left\{ \frac{dx}{\left( x^{2} + 1 \right)} - \int\frac{dx}{\left( x^{2} + 4 \right)} \right\} \\&= \frac{1}{3}\int\left\{ \frac{dx}{\left( 1 + x^{2} \right)} - \frac{dx}{\left( 1 + \frac{x^{2}}{4} \right)} \right\} \\&= \frac{1}{3}\left\{ 2\tan^{- 1}\frac{x}{2} - \tan^{- 1}x \right\} .\end{aligned}
$$

Or thus,

$$
\begin{aligned}\int_{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}}^{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}} &=\frac{1}{3}\int_{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}}^{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}} \\&=\frac{1}{3}\int_{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}}^{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}} \\&=\frac{1}{3}\int_{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}}^{\frac{1}{\left(x^{2}+1\right)\left(x^{2}+4\right)}} \\&=\frac{1}{3}\left(2\tan^{-1}\frac{x}{2}-\tan^{-1}x\right)\end{aligned}
